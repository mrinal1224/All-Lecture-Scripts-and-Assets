**Title:** Implementing JSON Web Token (JWT) Authentication in Express

**Problem Description:** 

You are tasked to implement JWT authentication in an Express application. The application has two routes: `/login` and `/secret`.

The `/login` route should accept a POST request with a JSON body containing `username` and `password`. If the username is "admin" and the password is "password", it should respond with a JWT token. The JWT token should be signed with the secret key "secretKey" and have a payload containing `username: "admin"`.

The `/secret` route should accept a GET request with a JWT token in the `Authorization` header. If the token is valid, it should respond with the message "Welcome to the secret area, [username]!". If the token is not valid, it should respond with the status code 401 and the message "Unauthorized".

**Code Stub:**

```javascript
const express = require('express');
const jwt = require('jsonwebtoken');

const app = express();
app.use(express.json());

app.post('/login', (req, res) => {
  // Your code goes here
});

app.get('/secret', (req, res) => {
  // Your code goes here
});

app.listen(3000, () => console.log('Server started on port 3000'));
```

**Solution:**

```javascript
app.post('/login', (req, res) => {
  const { username, password } = req.body;
  if (username === 'admin' && password === 'password') {
    const token = jwt

.sign

({ username: 'admin' }, 'secretKey');
    res.json({ token });
  } else {
    res.status(401).json({ message: 'Invalid credentials' });
  }
});

app.get('/secret', (req, res) => {
  const authHeader = req.headers.authorization;
  if (authHeader) {
    const token = authHeader.split(' ')[1];
    jwt.verify(token, 'secretKey', (err, user) => {
      if (err) {
        return res.sendStatus(403);
      }
      res.send(`Welcome to the secret area, ${user.username}!`);
    });
  } else {
    res.sendStatus(401);
  }
});
```

Note: This solution assumes that the 'express' and 'jsonwebtoken' modules are installed in your project. If not, you can install them using `npm install express jsonwebtoken`.
